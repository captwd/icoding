## 3.堆元素插入
### 参考答案
```c
#include <stdio.h>
#include <stdlib.h>
#include "minbinheap.h"

bool heap_insert_value(PMinHeap pq, int value){
    if(pq->heap_size==pq->capacity)
    {
        return false;
    }
    int i=pq->heap_size;  //最后的结点
    pq->heap_size++;
    pq->heap_array[i].value=value;
    pq->heap_array[i].otherInfo.i=0;     //这里随意赋值
    pq->heap_array[i].otherInfo.j=0;
    while(i!=0 && value<pq->heap_array[parent(i)].value)   //比父小就交换上去
    {
        swap_node(&pq->heap_array[i],&pq->heap_array[parent(i)]);
        i=parent(i);
    }
    return true;
}
```
### 思路
堆每一行并不是一个有序排列，因此不能单纯地像冒泡排序那样比大小，交换顺序

堆地顺序体现在上下间的大小，因此是父子结点的比较，而不是兄弟结点的比较

这里插入选择在末尾插入，然后一层层往上比较

## 4.堆化

### 参考答案
```c
#include <stdio.h>
#include <stdlib.h>
#include "minbinheap.h"

void min_heapify(PMinHeap pq, int i){
    int small=i;
    int j=left(i);
    int k=right(i);
    if(j < pq->heap_size && pq->heap_array[small].value>pq->heap_array[j].value)//不加
    {
        small=j;
    }
    if(k < pq->heap_size && pq->heap_array[small].value>pq->heap_array[k].value)  //这个size比较如果不比会出错
    {
        small=k;
    }                           //该结点与其左右子结点进行值比较，选择三个中最小的重新做父亲
    if(small!=i)
    {
        swap_node(&pq->heap_array[small],&pq->heap_array[i]);
        min_heapify(pq,small);
    }
}
```
### 思路
这题话都不讲清楚，理解了半天，简单说就是，现在有一个已经很完美的最小堆，然后发现中间有一个结点
太大了，不符合这个位置，除了这一个结点出问题，其他都没有问题，就需要向下找到它应该在的位置
```
               [数值: 100] 
               /         \
             /             \
          (左)             (右)
       [数值: 10]          [数值: 15]
        /     \             /     \
      20       30         40       50
```

## 5.数组合并
最后一题绝杀

### 参考答案
```c
#include<stdio.h>
#include<stdlib.h>

typedef struct _otherInfo {  
    int i;  
    int j;  
} OtherInfo;  
  
typedef struct _minHeapNode {  
    int value;  
    OtherInfo otherInfo;  
} MinHeapNode, *PMinHeapNode;  
  
typedef struct _minPQ {  
    PMinHeapNode heap_array; 
    int heap_size; 
    int capacity;  
} MinHeap, *PMinHeap;  

int parent(int i) { 
    return (i - 1) / 2; 
} 

int left(int i) { 
    return i * 2 + 1; 
} 

int right(int i) { 
    return i * 2 + 2; 
} 

void swap_node(MinHeapNode *x, MinHeapNode *y) { 
    MinHeapNode temp = *x; 
    *x = *y; 
    *y = temp; 
} 

void min_heapify(PMinHeap pq, int i) { 
    int small = i; 
    int j = left(i); 
    int k = right(i); 

    if (j < pq->heap_size && pq->heap_array[small].value > pq->heap_array[j].value) { 
        small = j; 
    } 
    if (k < pq->heap_size && pq->heap_array[small].value > pq->heap_array[k].value) { 
        small = k; 
    } 
    if (small != i) { 
        swap_node(&pq->heap_array[small], &pq->heap_array[i]); 
        min_heapify(pq, small); 
    } 
} 
   //把前面定义的函数，结构体抄过来
void merge_arrays(const int* arr, int n, int k, int* output) { 
    if (n == 0 || k == 0) return; 

    MinHeap pq; 
    pq.capacity = n; 
    pq.heap_size = n;    
    pq.heap_array = (MinHeapNode*)malloc(n * sizeof(MinHeapNode)); //只需要比较每个数组最前面的那个，即最小的那个
    
    for (int i = 0; i < n; i++) { 
        pq.heap_array[i].value = arr[i * k];   //按行排列，0开头对应0，1开头对应4
        pq.heap_array[i].otherInfo.i = i;    //第i个数组
        pq.heap_array[i].otherInfo.j = 1;    //下一个替补
    } 

    for (int i = (n - 2) / 2; i >= 0; i--) { 
        min_heapify(&pq, i);    从最后一个不断往上走进行排序
    } 

    for (int count = 0; count < n * k; count++) { 
        MinHeapNode root = pq.heap_array[0];   把最小的先塞进去
        output[count] = root.value; 
        
        int array_index = root.otherInfo.i;       //找到是哪个数组
        int next_element_index = root.otherInfo.j;   //下一个替补

        if (next_element_index < k) { 
            pq.heap_array[0].value = arr[array_index * k + next_element_index]; 
            pq.heap_array[0].otherInfo.i = array_index; 
            pq.heap_array[0].otherInfo.j = next_element_index + 1; 
        } else { 
            pq.heap_array[0] = pq.heap_array[pq.heap_size - 1];   //第一个数组全部排进去了，把最下面的拿过来排序
            pq.heap_size--; 
        } 

        if (pq.heap_size > 0) { 
            min_heapify(&pq, 0);    //进行堆化，找到合适位置
        } 
    } 
    free(pq.heap_array);     //都已经排好了，不需要这块内存了
}
```

### 思路
这题用堆时间复杂度最小，但是要自己把前面定义的函数超过来
