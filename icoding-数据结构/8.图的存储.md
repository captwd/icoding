## 1.邻接矩阵

试在邻接矩阵存储结构上实现图的基本操作 matrix_insert_vertex 和matrix_insert_arc，相关定义如下：
```c
typedef int VertexType;

typedef enum{
    DG, UDG
}GraphType;

typedef struct{
    VertexType vertex[MAX_VERTEX_NUM]; //顶点向量
    int arcs[MAX_VERTEX_NUM][MAX_VERTEX_NUM]; //邻接矩阵
    int vexnum, arcnum;   //图的当前顶点数和弧数
    GraphType type;     //图的种类标志
}MatrixGraph;

int matrix_locate_vertex(MatrixGraph *MG, VertexType vex); //返回顶点 v 在vertex数组中的下标，如果v不存在，返回-1
bool matrix_insert_vertex(MatrixGraph *G, VertexType v);
bool matrix_insert_arc(MatrixGraph *G, VertexType v, VertexType w);
```
当成功插入顶点或边时，函数返回true，否则（如顶点或边已存在、插入边时顶点v或w不存在）返回false。

### 参考答案
```c
#include <stdio.h>
#include "graph.h" // 请不要删除，否则检查不通过

bool matrix_insert_vertex(MatrixGraph *G, VertexType v){
    if(G->vexnum >=MAX_VERTEX_NUM)
    {
        return false;
    }
    if(matrix_locate_vertex(G,v)!=-1)
    {
        return false;
    }
    int new_vertex=G->vexnum;
    G->vertex[new_vertex]=v;
    for(int i=0;i<MAX_VERTEX_NUM;i++)
    {
        G->arcs[i][new_vertex]=0;   //数据初始化
        G->arcs[new_vertex][i]=0;
    }
    G->vexnum++;
    return true;
}

bool matrix_insert_arc(MatrixGraph *G, VertexType v, VertexType w){
    int v_vertex=matrix_locate_vertex(G,v);
    int w_vertex=matrix_locate_vertex(G,w);
    if(v_vertex==-1)
    {
        return false;
    }
    if(w_vertex==-1)
    {
        return false;
    }
    if(G->arcs[v_vertex][w_vertex]==1)
    {
        return false;
    }
    if(G->type==DG)      //区分有向图和无向图
    {
        G->arcs[v_vertex][w_vertex]=1;
    }else{
        G->arcs[v_vertex][w_vertex]=1;
        G->arcs[w_vertex][v_vertex]=1;
    }
    G->arcnum++;
    return true;
}
```
## 2.邻接表1

虽然代码看着多，但其实比较正常

试在邻接表存储结构上实现图的基本操作 insert_vertex 和 insert_arc，相关定义如下：
```c
typedef int VertexType;

typedef enum{
    DG, UDG
}GraphType;

typedef struct ArcNode
{
    int adjvex;
    InfoPtr *info;
    struct ArcNode *nextarc;

}ArcNode;

typedef struct VNode
{
    VertexType data;
    ArcNode *firstarc;
}VNode;
typedef struct
{
    VNode vertex[MAX_VERTEX_NUM];
    int vexnum, arcnum;
    GraphType type;
}ListGraph;

int locate_vertex(ListGraph* G, VertexType v); //返回顶点 v 在vertex数组中的下标，如果v不存在，返回-1
bool insert_vertex(ListGraph *G, VertexType v);
bool insert_arc(ListGraph *G, VertexType v, VertexType w);
```
当成功插入顶点或边时，函数返回true，否则（如顶点或边已存在、插入边时顶点v或w不存在）返回false。

### 参考答案
```c
#include <stdio.h>
#include "graph.h" //请勿删除，否则检查不通过

bool insert_vertex(ListGraph *G, VertexType v){   //插入结点
    if(G->vexnum>=MAX_VERTEX_NUM || locate_vertex(G,v)!=-1)
    {
        return false;
    }
    int new_num=G->vexnum;
    G->vertex[new_num].data=v;
    G->vertex[new_num].firstarc=NULL;
    G->vexnum++;
    return true;
}


bool insert_arc(ListGraph *G, VertexType v, VertexType w){
    if(G->vexnum>=MAX_VERTEX_NUM || locate_vertex(G,v)==-1 || locate_vertex(G,w)==-1)
    {
        return false;
    }
    int num1=locate_vertex(G,v);
    int num2=locate_vertex(G,w);   //找到对应的位置
    ArcNode *n=(ArcNode*)malloc(sizeof(ArcNode));
    n=G->vertex[num1].firstarc;
    while(n!=NULL)
    {
        if(n->adjvex==num2)
        {
            return false;
        }
        n=n->nextarc;
    }
    ArcNode *p=(ArcNode*)malloc(sizeof(ArcNode));
    p->adjvex=num2;
    p->info=NULL;
    p->nextarc=G->vertex[num1].firstarc;
    G->vertex[num1].firstarc=p;
    if(G->type!=DG)    //再多一条边
    {
        ArcNode *q=(ArcNode*)malloc(sizeof(ArcNode*));
        q->adjvex=num1;
        q->info=NULL;
        q->nextarc=G->vertex[num2].firstarc;
        G->vertex[num2].firstarc=q;
        G->arcnum++;
    }
    G->arcnum++;
    return true;
}
```

## 2.邻接表2

试在邻接表存储结构上实现图的基本操作 del_vertex，相关定义如下：
```c
typedef int VertexType;

typedef enum{
    DG, UDG
}GraphType;

typedef struct ArcNode{
    int adjvex;
    InfoPtr *info;
    struct ArcNode *nextarc;
}ArcNode;

typedef struct VNode{
    VertexType data;
    ArcNode *firstarc;
}VNode;
typedef struct{
    VNode vertex[MAX_VERTEX_NUM];
    int vexnum, arcnum;
    GraphType type;
}ListGraph;

int locate_vertex(ListGraph *G, VertexType v); //返回顶点 v 在vertex数组中的下标，如果v不存在，返回-1
bool del_vertex(ListGraph *G, VertexType v); //删除顶点 v
```
当成功删除顶点或边时，函数返回true，否则（如顶点或边不存在、删除边时顶点v或w不存在）返回false。

### 参考答案
```c
#include <stdio.h>
#include "graph.h" //请勿删除，否则检查不通过
#include <stdlib.h>

bool del_vertex(ListGraph *G, VertexType v){ 
    int num1=locate_vertex(G,v);
    if(num1==-1)
    {
        return false;
    }
    ArcNode *p=G->vertex[num1].firstarc;  // 删除该顶点引出的所有边
    int num_del=0;  //记录删掉的边的个数
    while(p!=NULL)
    {
        ArcNode *temp=p;
        p=p->nextarc;
        free(temp);
        num_del++;
    }
    
    for(int i=num1;i<G->vexnum-1;i++)  //数组整体前移覆盖
    {
        G->vertex[i]=G->vertex[i+1];
    }
    
    for(int i=0;i<G->vexnum-1;i++)  //遍历剩下的所有点，处理连带关系
    {
        ArcNode *q =G->vertex[i].firstarc;
        ArcNode *prev=NULL;
        while(q!=NULL)
        {
            if(q->adjvex==num1)   // 这条边指向被删的城，彻底删除它
            {
                ArcNode *to_delete=q;
                if(prev==NULL)
                {
                    G->vertex[i].firstarc=q->nextarc;
                }
                else{
                    prev->nextarc=q->nextarc;
                }
                free(to_delete);
                num_del++;
                break;
            }
            prev=q;
            q=q->nextarc;
        }
    }
    G->vexnum--;
    G->arcnum=G->arcnum-num_del;
    return true;
}
```
